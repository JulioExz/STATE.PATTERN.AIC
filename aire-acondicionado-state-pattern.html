/**
 * PATRÓN STATE — Contexto
 *
 * El Contexto mantiene una referencia al estado
 * actual y delega en él todo el comportamiento.
 * No necesita saber QUÉ hace cada estado.
 */

class AirConditionerContext {

  constructor() {
    // El contexto empieza en estado Apagado
    this.currentState = new OffState(this);
    this.ui = new AcUI();
  }

  /**
   * transitionTo — único punto de cambio de estado.
   * Los estados llaman a este método para cambiar
   * el comportamiento del contexto en tiempo de
   * ejecución (sin condicionales).
   */
  transitionTo(state) {
    this.currentState = state;
    this.currentState.enter();
  }

  /**
   * request — punto de entrada desde la UI.
   * El contexto simplemente pregunta al estado
   * actual qué hacer con la solicitud.
   */
  request(action, btn) {
    this.currentState.handle(action, btn);
  }
}

// Instancia global del contexto (el "AC")
const ctx = new AirConditionerContext();

/**
 * PATRÓN STATE — Interfaz base y Estados Concretos
 *
 * Cada estado encapsula su propio comportamiento.
 * Todos implementan enter() y handle().
 */

// ── Interfaz / clase base ─────────────────────
class AcState {
  constructor(context) {
    this.ctx = context;
  }
  enter()  { /* se ejecuta al ingresar al estado */ }
  handle() { /* procesa la solicitud de la UI   */ }
}

// ── Estado: Apagado ───────────────────────────
class OffState extends AcState {
  enter() {
    this.ctx.ui.render({
      unitClass: 'state-off',
      screen:    '— — —',
      screenCls: 'off',
      led:       '',
      airType:   null,
      label:     'Apagado'
    });
  }
  handle(action, btn) {
    // Desde apagado solo se puede encender o poner en reposo
    if (action === 'on')
      this.ctx.transitionTo(new OnState(this.ctx), btn);
    else if (action === 'standby')
      this.ctx.transitionTo(new StandbyState(this.ctx), btn);
  }
}

// ── Estado: Encendido ─────────────────────────
class OnState extends AcState {
  enter() {
    this.ctx.ui.render({
      unitClass: 'state-on',
      screen:    '24 °C',
      screenCls: 'on',
      led:       'active',
      airType:   'fresh',
      label:     'Encendido'
    });
  }
  handle(action, btn) {
    const map = {
      off:     OffState,
      cold:    ColdState,
      heat:    HeatState,
      standby: StandbyState
    };
    if (map[action])
      this.ctx.transitionTo(new map[action](this.ctx), btn);
  }
}

/**
 * UI / Render
 *
 * AcUI es la capa de presentación. Recibe los
 * datos del estado activo y actualiza el DOM.
 * No contiene lógica de negocio.
 */

class AcUI {

  constructor() {
    this.unit    = document.getElementById('ac-unit');
    this.led     = document.getElementById('ac-led');
    this.screen  = document.getElementById('ac-screen');
    this.airflow = document.getElementById('airflow');
    this.label   = document.getElementById('state-label');
    this._particleTimer = null;
  }

  /**
   * render — recibe la descripción del estado y
   * actualiza todos los elementos visuales.
   */
  render({ unitClass, screen, screenCls, led, airType, label }) {

    // 1. Clases en la unidad (color, borde)
    this.unit.className = `ac-unit ${unitClass}`;

    // 2. Pantalla digital
    this.screen.textContent = screen;
    this.screen.className   = `ac-screen ${screenCls}`;

    // 3. LED de estado
    this.led.className = `ac-led ${led}`;

    // 4. Etiqueta descriptiva
    this.label.textContent = `Estado actual: ${label}`;

    // 5. Partículas de flujo de aire
    this._clearParticles();
    if (airType) this._spawnParticles(airType);

    // 6. Resaltar botón activo
    document.querySelectorAll('.state-btn').forEach(
      b => b.classList.remove('active')
    );
  }

  _clearParticles() {
    clearInterval(this._particleTimer);
    this.airflow.innerHTML = '';
  }

  _spawnParticles(type) {
    const positions = [20, 55, 90, 125, 160, 200, 238];
    const delays    = [0, 0.3, 0.6, 0.1, 0.5, 0.2, 0.7];
    positions.forEach((x, i) => {
      const p = document.createElement('div');
      p.className           = `particle ${type}`;
      p.style.left          = x + 'px';
      p.style.animationDelay= delays[i] + 's';
      this.airflow.appendChild(p);
    });
  }
}

// Activa la pestaña de código seleccionada
function showTab(name) {
  document.querySelectorAll('.code-block').forEach(
    el => el.classList.remove('visible')
  );
  document.querySelectorAll('.code-tab').forEach(
    el => el.classList.remove('active')
  );
  document.getElementById(`tab-${name}`)
           .classList.add('visible');
  event.target.classList.add('active');
}
